import { createHash, timingSafeEqual } from 'node:crypto';
import { createAuthThrottle } from './authThrottle.js';
import { clientKeyFrom } from './clientKey.js';

/** Hosts that only this machine can reach. Anything else is a public bind. */
const LOOPBACK = new Set(['localhost', '127.0.0.1', '::1']);

/**
 * Does a server bound to `host` have to be behind the gate?
 *
 * Derived from the host actually bound, not from an env var: `vite preview
 * --host 0.0.0.0` binds publicly while leaving HOST unset, so reading the
 * environment alone left the CLI path serving everything unauthenticated.
 * Vite's `--host` with no value resolves to `true`, which is also public.
 */
export function requiresGate(host) {
  if (host === true) return true;
  if (!host || typeof host !== 'string') return false;
  return !LOOPBACK.has(host.trim());
}

/**
 * HTTP Basic gate for the whole app, dev server and preview alike.
 *
 * Deployed, this app hands out real money: the browser bundle carries the
 * Google Maps key, and /api/realtime/token mints OpenAI sessions against the
 * server's own credentials. Anyone who can load the page can spend both, so a
 * public deployment has to be closed by default.
 *
 * Unset credentials leave the gate uninstalled rather than open-with-a-warning,
 * so localhost keeps working untouched; a half-configured gate (user but no
 * password) also stays shut rather than authenticating against an empty secret.
 *
 * Lives in src/ because the unit runner discovers src/**\/*.test.mjs only.
 */

/**
 * Compare without leaking length or position through timing.
 *
 * Digesting first is what makes that true. Comparing the raw bytes forces an
 * early return whenever the lengths differ, which leaks the exact byte length
 * of the username and password to anyone timing responses; SHA-256 makes both
 * sides 32 bytes, so every comparison does identical work.
 */
function safeEqual(a, b) {
  const digest = (value) => createHash('sha256').update(String(value), 'utf8').digest();
  return timingSafeEqual(digest(a), digest(b));
}

/**
 * @param {{user?: string, password?: string, realm?: string, required?: boolean}} credentials
 *   `required` marks a deployment that must never serve unprotected. Without it
 *   missing credentials merely uninstall the gate, which is right for localhost
 *   and a silent failure open anywhere public.
 * @returns {import('vite').Plugin}
 */
export function basicAuthGate({ user, password, realm = "God's Eye View", required = false } = {}) {
  const armed = Boolean(user && password);
  // Shared across both hooks so dev and preview cannot be used as two
  // independent guessing budgets.
  const throttle = createAuthThrottle();

  // A public bind detected at hook time also requires the gate, so the caller
  // cannot forget to pass `required` on the CLI path.
  function install(middlewares, boundPublicly = false) {
    if (!armed) {
      if (!required && !boundPublicly) return;
      // Refuse to serve rather than serve unguarded: this process hands out the
      // Google Maps key in its bundle and mints OpenAI sessions on request, so
      // one misnamed variable would otherwise publish both. 503, not 401 -
      // nothing the caller can send would help. The message deliberately names
      // no variable, since it is returned to the public internet.
      middlewares.use((req, res) => {
        res.statusCode = 503;
        res.setHeader('Cache-Control', 'no-store');
        res.end('Service unavailable: access control is not configured.');
      });
      return;
    }
    middlewares.use((req, res, next) => {
      // Behind Cloud Run the socket peer is the front end, so the same hop
      // count the spend limiters use is what makes this per-caller.
      const who = clientKeyFrom(req, process.env.GEV_TRUSTED_PROXY_HOPS);
      const wait = throttle.blockedFor(who);
      if (wait > 0) {
        // 429 before any comparison: a caller in cooldown gets no further
        // signal, and a guess costs them a refusal rather than a test.
        res.statusCode = 429;
        res.setHeader('Retry-After', String(wait));
        res.setHeader('Cache-Control', 'no-store');
        res.end('Too many failed attempts');
        return;
      }
      const header = req.headers?.authorization || '';
      // RFC 7235 allows optional whitespace between the scheme and the
      // credentials, so `Basic  <token>` is valid. Splitting on one space left
      // `encoded` empty and rejected a conforming client.
      const [scheme, encoded] = header.trim().split(/\s+/);

      // RFC 7235 makes the auth-scheme case-insensitive; matching only "Basic"
      // locks out a conforming client rather than stopping an attacker.
      if (scheme?.toLowerCase() === 'basic' && encoded) {
        // Buffer.from never throws on bad base64 - it drops invalid bytes - so
        // the separator check below is what actually rejects malformed input.
        const decoded = Buffer.from(encoded, 'base64').toString('utf8');
        const separator = decoded.indexOf(':');
        if (separator !== -1) {
          const okUser = safeEqual(decoded.slice(0, separator), user);
          const okPassword = safeEqual(decoded.slice(separator + 1), password);
          // Both are evaluated before branching so a wrong username costs the
          // same as a wrong password.
          if (okUser && okPassword) {
            throttle.recordSuccess(who);
            next();
            return;
          }
        }
      }

      throttle.recordFailure(who);
      res.statusCode = 401;
      res.setHeader('WWW-Authenticate', `Basic realm="${realm}", charset="UTF-8"`);
      res.end('Authentication required');
    });
  }

  return {
    name: 'basic-auth-gate',
    configureServer(server) {
      install(server.middlewares, requiresGate(server?.config?.server?.host));
    },
    configurePreviewServer(server) {
      install(server.middlewares, requiresGate(server?.config?.preview?.host));
    },
  };
}

==== FILE: src/server/authThrottle.js ====
/**
 * Rate-limit failed authentication attempts against the gate.
 *
 * Without this the gate answers every wrong password with an immediate 401 and
 * keeps no count, so guessing is unlimited - confirmed externally at 1,000
 * consecutive attempts, none throttled. The app's own spend limiters cannot
 * cover this: the gate runs ahead of them, so a guess never reaches one.
 *
 * The password is long enough that guessing it is not the realistic risk. The
 * cost is: on Cloud Run every attempt is a billable request, so an unthrottled
 * gate makes the only control on the app an unmetered one.
 *
 * The counter must not become the next problem. An attacker rotating source
 * addresses would grow the map without bound, so tracking is capped and evicts
 * least-recently-seen - which is why a Map is used directly: its insertion
 * order gives LRU eviction with no extra structure.
 *
 * Lives in src/ because the unit runner discovers src/**\/*.test.mjs only.
 */

/**
 * @param {{maxFailures?: number, windowMs?: number, cooldownMs?: number,
 *          maxTracked?: number, now?: () => number}} options
 */
export function createAuthThrottle({
  maxFailures = 8,
  windowMs = 5 * 60_000,
  cooldownMs = 5 * 60_000,
  maxTracked = 5_000,
  now = Date.now,
} = {}) {
  /** @type {Map<string, {count: number, first: number, blockedUntil: number}>} */
  const clients = new Map();

  /** Re-insert so the Map's insertion order stays least-recently-seen first. */
  function touch(key, entry) {
    clients.delete(key);
    clients.set(key, entry);
    while (clients.size > maxTracked) {
      clients.delete(clients.keys().next().value);
    }
  }

  return {
    /** Seconds the caller must wait, or 0 if it may proceed. */
    blockedFor(key) {
      const entry = clients.get(key);
      if (!entry) return 0;
      const t = now();
      if (entry.blockedUntil > t) return Math.ceil((entry.blockedUntil - t) / 1000);
      // The cooldown has run out, or the window lapsed with room to spare.
      if (entry.blockedUntil) clients.delete(key);
      return 0;
    },

    recordFailure(key) {
      const t = now();
      const entry = clients.get(key);
      // A first failure, or one arriving after the window closed, starts fresh
      // rather than inheriting a stale count.
      if (!entry || t - entry.first > windowMs) {
        touch(key, { count: 1, first: t, blockedUntil: 0 });
        return;
      }
      entry.count += 1;
      if (entry.count >= maxFailures) entry.blockedUntil = t + cooldownMs;
      touch(key, entry);
    },

    /** Proving you know the password clears your record. */
    recordSuccess(key) {
      clients.delete(key);
    },

    size() {
      return clients.size;
    },
  };
}

==== FILE: src/server/clientKey.js ====
/**
 * Identify the caller for rate limiting, correctly behind a known proxy.
 *
 * Upstream reads `req.socket.remoteAddress` and deliberately ignores
 * X-Forwarded-For, because a client can rotate that header to mint fresh quota.
 * That is right for a localhost dev proxy, where the socket peer is the client.
 *
 * Deployed behind Cloud Run it is wrong in the other direction: every peer is
 * Google's front end, so all callers collapse into one bucket and a single
 * abuser exhausts the shared OpenAI and Google budget for everyone.
 *
 * The safe reading is from the RIGHT. A proxy appends the address it saw, so
 * with N trusted proxies the entry at `length - N` is the last one an attacker
 * could not write; anything further left is attacker-supplied and ignored. With
 * zero trusted hops the header is not consulted at all, preserving upstream's
 * behaviour exactly - which is why the default is 0 and deployment opts in.
 *
 * Lives in src/ because the unit runner discovers src/**\/*.test.mjs only.
 */

/**
 * Fold equivalent spellings of one address together.
 *
 * `1.2.3.4`, `1.2.3.4:443` and `::ffff:1.2.3.4` are the same client, and left
 * distinct they are three separate rate-limit buckets - three times the quota
 * for anyone who varies the spelling. Real IPv6 is left alone: it is only the
 * IPv4-mapped form and a trailing IPv4 port that are rewritten.
 */
function canonicalizeAddress(value) {
  let address = value.trim().toLowerCase();
  if (address.startsWith('::ffff:') && address.includes('.')) address = address.slice('::ffff:'.length);
  // Strip a port only from IPv4 - a bare colon in IPv6 is part of the address.
  if (/^\d{1,3}(\.\d{1,3}){3}:\d+$/.test(address)) address = address.split(':')[0];
  return address;
}

/** Hops must be a small non-negative integer; anything else means "trust nothing". */
function normalizeHops(value) {
  const hops = Math.trunc(Number(value));
  if (!Number.isFinite(hops) || hops <= 0) return 0;
  return Math.min(hops, 8);
}

/**
 * @param {{headers?: Record<string, string|string[]>, socket?: {remoteAddress?: string}}} req
 * @param {number|string} trustedProxyHops
 * @returns {string}
 */
export function clientKeyFrom(req, trustedProxyHops = 0) {
  const rawSocket = req?.socket?.remoteAddress;
  const socketKey = rawSocket ? canonicalizeAddress(String(rawSocket)) : 'local';
  const hops = normalizeHops(trustedProxyHops);
  if (hops === 0) return socketKey;

  const raw = req?.headers?.['x-forwarded-for'];
  const parts = (Array.isArray(raw) ? raw.join(',') : String(raw ?? ''))
    .split(',')
    .map((part) => part.trim())
    .filter(Boolean);

  // Too few entries means the request did not traverse the proxies we were told
  // to trust, so there is no entry we can attribute to one. Falling back beats
  // reading the left edge, which is exactly the value an attacker controls.
  if (parts.length < hops) return socketKey;

  const selected = parts[parts.length - hops];
  return selected ? canonicalizeAddress(selected) : socketKey;
}

==== FILE: src/server/processGuards.js ====
/**
 * Keep the server alive through faults, and shut it down cleanly on request.
 *
 * Cloud Run runs one process per instance and restarts it when it exits, so a
 * single unhandled rejection anywhere across 19 proxy routes takes the whole
 * app down - and `restartPolicyType: on-failure` then hides the cause behind a
 * restart loop. Node's default for an unhandled rejection is to terminate, so
 * "one upstream returned something unexpected" and "the service is down" are
 * the same event until this is installed.
 *
 * SIGTERM matters for the opposite reason: Cloud Run sends it before every
 * scale-down and every deploy. Ignoring it drops in-flight requests during a
 * routine release, which reads to a user as random failures at deploy time.
 *
 * Lives in src/ because the unit runner discovers src/**\/*.test.mjs only.
 */

const INSTALLED = Symbol.for('gev.processGuards.installed');

/**
 * @param {{proc?: NodeJS.Process, log?: (msg: string) => void, onShutdown?: () => Promise<void> | void}} options
 */
export function installProcessGuards({ proc = process, log = console.error, onShutdown } = {}) {
  // Vite restarts the dev server in-process on config change while module state
  // survives; without this each reload stacks another set of handlers.
  if (proc[INSTALLED]) return proc;
  proc[INSTALLED] = true;

  const describe = (error) => (error instanceof Error ? `${error.message}\n${error.stack}` : String(error));

  proc.on('unhandledRejection', (reason) => {
    log(`[gev] unhandled rejection (kept alive): ${describe(reason)}`);
  });

  proc.on('uncaughtException', (error) => {
    // Deliberately non-fatal to match the rejection case: one bad request
    // handler should degrade a layer, not end the instance.
    log(`[gev] uncaught exception (kept alive): ${describe(error)}`);
  });

  const shutdown = async (signal) => {
    log(`[gev] ${signal} received, shutting down`);
    try {
      await onShutdown?.();
    } catch (error) {
      // Never let a failed cleanup wedge the shutdown - Cloud Run SIGKILLs
      // shortly after SIGTERM regardless.
      log(`[gev] shutdown hook failed: ${describe(error)}`);
    }
    proc.exitCode = 0;
  };

  proc.on('SIGTERM', () => shutdown('SIGTERM'));
  proc.on('SIGINT', () => shutdown('SIGINT'));

  return proc;
}

==== FILE: src/server/immutableAssets.js ====
/**
 * Cache content-hashed build assets immutably.
 *
 * `vite preview` - which serves the deployment, because the API surface is
 * plugin middleware - sends `cache-control: no-cache` for everything. On a
 * hashed filename that is pure waste: the browser revalidates several MB of
 * JS and JSON it already holds on every visit. A hashed name cannot change
 * contents without changing the URL, so it is safe to freeze for a year.
 *
 * index.html is deliberately excluded: it is what points at the current
 * hashes, so freezing it would pin visitors to a stale build.
 *
 * Lives in src/ because the unit runner discovers src/**\/*.test.mjs only.
 */

export const IMMUTABLE_CACHE_CONTROL = 'public, max-age=31536000, immutable';

/** Vite emits `name-<hash>.ext`; the hash is base64url-ish and 8+ chars. */
const HASHED_ASSET = /^\/assets\/.+-[A-Za-z0-9_-]{8,}\.[a-z0-9]+$/i;

export function immutableAssets() {
  function install(middlewares) {
    middlewares.use((req, res, next) => {
      const pathname = (req.url || '').split('?')[0];
      if (HASHED_ASSET.test(pathname)) {
        res.setHeader('Cache-Control', IMMUTABLE_CACHE_CONTROL);
        // The static handler sets its own Cache-Control after this returns, so
        // pin ours: without this the header is silently downgraded again.
        const setHeader = res.setHeader.bind(res);
        res.setHeader = (name, value) =>
          (String(name).toLowerCase() === 'cache-control' ? res : setHeader(name, value));
      }
      next();
    });
  }

  return {
    name: 'immutable-assets',
    configureServer(server) {
      install(server.middlewares);
    },
    configurePreviewServer(server) {
      install(server.middlewares);
    },
  };
}

==== FILE: src/annotations/geocodeNameMatch.js ====
/**
 * Did the geocoder echo back the name the user actually gave?
 *
 * The resolver rejects a geocoded point that sits far from the current view,
 * which is right when a vague local ask resolves to a distant wrong thing —
 * "the Texas Capitol" coming back as the STATE of Texas. It is wrong when the
 * user named a place in full and the geocoder returned exactly that place:
 * marking LAX from a camera over Texas is not a mis-match, it is the request.
 * That case was rejected at 1991km against a 50km limit and reported to the
 * user as "nothing could be marked".
 *
 * A name echo alone is not enough: "Empire State" geocodes to an exact echo
 * while meaning the Building, and the resolver deliberately keeps that on the
 * near-view recovery path. So the bypass also requires the geocoder's own type
 * to name an unambiguous transport hub - the class the launch demo marks and
 * the class whose full name admits no second reading.
 */

/** Geocode types whose named facility is unambiguous wherever the camera is. */
const UNAMBIGUOUS_TYPES = new Set([
  'airport', 'international_airport', 'transit_station',
  'train_station', 'bus_station', 'subway_station', 'ferry_terminal',
]);

/** Lowercase, drop a leading article, strip punctuation and parenthetical codes. */
function normalize(value) {
  return String(value ?? '')
    .toLowerCase()
    .replace(/\([^)]*\)/g, ' ')      // "(LAX)" is a code, not part of the name
    .replace(/^\s*the\s+/, ' ')
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

const tokens = (value) => normalize(value).split(' ').filter(Boolean);

/**
 * @param {string} query        what the user asked for
 * @param {string} primaryName  the geocoder's canonical name for what it found
 * @param {string[]} [types]    the geocoder's types for that result
 * @returns {boolean} true when the result is an unambiguous echo of the ask and
 *   may skip the proximity gate; false whenever the gate should still apply.
 */
export function geocodeNameConfidentlyMatches(query, primaryName, types = []) {
  if (!Array.isArray(types) || !types.some((t) => UNAMBIGUOUS_TYPES.has(String(t)))) return false;
  const askTokens = tokens(query);
  const gotTokens = tokens(primaryName);
  if (!askTokens.length || !gotTokens.length) return false;

  // An exact echo is unambiguous at any distance, including a single word.
  if (askTokens.join(' ') === gotTokens.join(' ')) return true;

  // Otherwise the canonical name has to account for the ask rather than a
  // fragment of it. One token out of several is the "Texas Capitol" → "Texas"
  // shape the gate is there to catch, so it stays gated.
  if (gotTokens.length < 2) return false;

  const ask = new Set(askTokens);
  return gotTokens.every((token) => ask.has(token));
}
